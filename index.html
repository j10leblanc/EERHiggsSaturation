<script>

// Scene Setup

const scene = new THREE.Scene();

scene.fog = new THREE.FogExp2(0x05070b, 0.035);


const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);

camera.position.set(0, 16, 22);


const renderer = new THREE.WebGLRenderer({ antialias: true });

renderer.setSize(window.innerWidth, window.innerHeight);

renderer.setPixelRatio(window.devicePixelRatio);

document.body.appendChild(renderer.domElement);


const controls = new THREE.OrbitControls(camera, renderer.domElement);

controls.enableDamping = true;

controls.dampingFactor = 0.05;


// Lighting

const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);

scene.add(ambientLight);


const pointLight = new THREE.PointLight(0x58a6ff, 2, 50);

pointLight.position.set(0, 12, 0);

scene.add(pointLight);


// 1. Dynamic Deformable Pseudomanifold Grid (Planar Action Space)

const gridDim = 20;

const planeGeo = new THREE.PlaneGeometry(gridDim, gridDim, 24, 24);

planeGeo.rotateX(-Math.PI / 2);


const planeMat = new THREE.MeshStandardMaterial({

    color: 0x0d1117,

    emissive: 0x1f6feb,

    emissiveIntensity: 0.15,

    wireframe: true,

    side: THREE.DoubleSide

});

const deformableGrid = new THREE.Mesh(planeGeo, planeMat);

scene.add(deformableGrid);


// 2. Topological Defect Node (Particle Traversing Substrate)

const nodeGeo = new THREE.SphereGeometry(0.6, 32, 32);

const nodeMat = new THREE.MeshStandardMaterial({ color: 0x79c0ff, emissive: 0x79c0ff, emissiveIntensity: 0.5 });

const defectNode = new THREE.Mesh(nodeGeo, nodeMat);

scene.add(defectNode);


// 3. Orthogonal Bulk Relief Vent ($\Upsilon$ leakage)

const ventGeo = new THREE.CylinderGeometry(0.06, 0.2, 5, 16);

const ventMat = new THREE.MeshBasicMaterial({ color: 0xd2a8ff, transparent: true, opacity: 0.0 });

const bulkVent = new THREE.Mesh(ventGeo, ventMat);

scene.add(bulkVent);


// Simulation Engine State

let clock = 0;

const statusText = document.getElementById('phaseStep');

const posAttribute = planeGeo.attributes.position;

// Store original baseline vertex heights

const initialY = [];

for (let i = 0; i < posAttribute.count; i++) {

    initialY.push(posAttribute.getY(i));

}


function animate() {

    requestAnimationFrame(animate);

    clock += 0.012;

    let cycle = clock % 10.0; // 10-second complete phase-locking cycle


    let nodeX = Math.sin(clock * 0.8) * 6.0;

    let nodeZ = Math.cos(clock * 0.5) * 4.0;

    defectNode.position.set(nodeX, 0, nodeZ);

    bulkVent.position.set(nodeX, 2.5, nodeZ);


    let saturationLevel = 0.0;


    if (cycle < 3.0) {

        // PHASE 1: Fluid Substrate (Zero/Low Capacity Friction - Massless regime)

        statusText.innerText = "1. Unconstrained Substrate: Fluid capacity exchange (Zero inertial drag)";

        nodeMat.color.setHex(0x79c0ff);

        nodeMat.emissive.setHex(0x79c0ff);

        ventMat.opacity = 0.0;

        saturationLevel = 0.1;

    } 

    else if (cycle < 5.0) {

        // PHASE 2: Phase-Locking Transition (Approaching LeBlanc Bound σ_LB)

        statusText.innerText = "2. Substrate Phase-Locking: Local capacity saturation building";

        nodeMat.color.setHex(0xff7b72);

        nodeMat.emissive.setHex(0xff7b72);

        ventMat.opacity = 0.4;

        saturationLevel = 0.5;

    } 

    else if (cycle < 8.5) {

        // PHASE 3: Saturated Lattice (High Capacity Friction -> Inertial Mass Acquired)

        statusText.innerText = "3. Saturated Lattice: High capacity friction (Inertial mass active & bulk venting)";

        nodeMat.color.setHex(0xffa657);

        nodeMat.emissive.setHex(0xffa657);

        ventMat.opacity = 0.8;

        saturationLevel = 1.0;

    } 

    else {

        // PHASE 4: Relaxation back to baseline fluidity

        statusText.innerText = "4. Lattice Relaxation: Capacity strain bleeding into bulk";

        ventMat.opacity = 0.2;

        saturationLevel = 0.2;

    }


    // Deform the 2D pseudomanifold grid vertices dynamically around the defect node

    for (let i = 0; i < posAttribute.count; i++) {

        let vx = posAttribute.getX(i);

        let vz = posAttribute.getZ(i);

        

        let dist = Math.sqrt(Math.pow(vx - nodeX, 2) + Math.pow(vz - nodeZ, 2));

        let deformation = Math.max(0, (2.5 - dist)) * saturationLevel * 0.8;

        

        posAttribute.setY(i, initialY[i] - deformation);

    }

    posAttribute.needsUpdate = true;


    controls.update();

    renderer.render(scene, camera);

}


animate();


window.addEventListener('resize', () => {

    camera.aspect = window.innerWidth / window.innerHeight;

    camera.updateProjectionMatrix();

    renderer.setSize(window.innerWidth, window.innerHeight);

});
